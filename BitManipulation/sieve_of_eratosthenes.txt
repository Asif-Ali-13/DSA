----------------------------------------------------------------------------------------

problem statement:
You are given an 2D array queries of dimension n*2.
The queries[i] represents a range from queries[i][0] to queries[i][1] 
(include the end points).
Return the count of prime numbers present in between each range in queries array.

----------------------------------------------------------------------------------------

approach:
    1. Use the Sieve of Eratosthenes to find all primes up to maxVal
    2. Create a prefix sum array to count primes up to each number
    3. Process each query to find the number of primes in the given range

----------------------------------------------------------------------------------------

code:

public ArrayList<Integer> primesInRange(ArrayList<int[]> queries) {
    if (queries == null || queries.isEmpty()) {
        return new ArrayList<>();
    }

    int maxVal = 0;
    for (int[] query : queries) {
        maxVal = Math.max(maxVal, query[1]);
    }

    // Step 1: Use the Sieve of Eratosthenes 

    boolean[] isPrime = new boolean[maxVal + 1];
    for (int i = 2; i <= maxVal; i++) {
        isPrime[i] = true;
    }
    for (int p = 2; p * p <= maxVal; p++) {
        if (isPrime[p]) {
            for (int i = p * p; i <= maxVal; i += p) {
                isPrime[i] = false;
            }
        }
    }

    // Step 2: 
    int[] primeCount = new int[maxVal + 1];
    for (int i = 1; i <= maxVal; i++) {
        primeCount[i] = primeCount[i - 1];
        if (isPrime[i]) {
            primeCount[i]++;
        }
    }

    // Step 3: 
    ArrayList<Integer> result = new ArrayList<>();
    for (int[] query : queries) {
        int start = query[0];
        int end = query[1];
        if (start == 0) {
            result.add(primeCount[end]);
        } else {
            result.add(primeCount[end] - primeCount[start - 1]);
        }
    }

    return result;
}

----------------------------------------------------------------------------------------

tc: O(N log (log N))
sc: O(N)
